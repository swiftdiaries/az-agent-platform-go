package maf_hitl

import (
	"context"
	"errors"
	"fmt"
	"os"
	"os/exec"
	"strings"
	"sync"
	"testing"
)

func TestConcurrentFreshProcessesOnlyOneClaimsReply(t *testing.T) {
	databaseURL := os.Getenv("MAF_SPIKE_DATABASE_URL")
	if databaseURL == "" {
		t.Skip("MAF_SPIKE_DATABASE_URL unavailable")
	}
	store, err := OpenStore(t.Context(), databaseURL)
	if err != nil {
		t.Fatal(err)
	}
	defer store.Close()
	if err := store.Ensure(t.Context()); err != nil {
		t.Fatal(err)
	}
	if err := store.Reset(t.Context()); err != nil {
		t.Fatal(err)
	}
	if err := store.Put(t.Context(), Pending{
		ThreadID: "race-thread", Kind: "approval", RequestID: "request-1",
		Principal: "alice", Session: []byte(`{}`),
	}); err != nil {
		t.Fatal(err)
	}

	start := make(chan struct{})
	results := make(chan string, 2)
	var wg sync.WaitGroup
	for i := range 2 {
		wg.Add(1)
		go func(i int) {
			defer wg.Done()
			<-start
			cmd := exec.Command(os.Args[0], "-test.run=TestStoreClaimHelperProcess")
			cmd.Env = append(os.Environ(),
				"MAF_SPIKE_HELPER=claim",
				fmt.Sprintf("MAF_SPIKE_CLAIMANT=process-%d", i),
			)
			out, err := cmd.CombinedOutput()
			if err != nil {
				results <- "process-error:" + err.Error() + ":" + string(out)
				return
			}
			results <- strings.TrimSpace(string(out))
		}(i)
	}
	close(start)
	wg.Wait()
	close(results)
	var claimed, duplicate int
	for result := range results {
		switch {
		case strings.Contains(result, "OUTCOME=claimed"):
			claimed++
		case strings.Contains(result, "OUTCOME=duplicate"):
			duplicate++
		default:
			t.Errorf("unexpected helper result: %s", result)
		}
	}
	if claimed != 1 || duplicate != 1 {
		t.Fatalf("claimed=%d duplicate=%d, want exactly one of each", claimed, duplicate)
	}
	if _, err := store.Claim(t.Context(), "race-thread", "request-1", "alice", "approval", "third-process"); !errors.Is(err, ErrAlreadyConsumed) {
		t.Fatalf("third claim error=%v, want ErrAlreadyConsumed", err)
	}
}

func TestStoreClaimHelperProcess(t *testing.T) {
	if os.Getenv("MAF_SPIKE_HELPER") != "claim" {
		return
	}
	store, err := OpenStore(context.Background(), os.Getenv("MAF_SPIKE_DATABASE_URL"))
	if err != nil {
		t.Fatal(err)
	}
	defer store.Close()
	_, err = store.Claim(context.Background(), "race-thread", "request-1", "alice", "approval", os.Getenv("MAF_SPIKE_CLAIMANT"))
	switch {
	case err == nil:
		fmt.Println("OUTCOME=claimed")
	case errors.Is(err, ErrAlreadyConsumed):
		fmt.Println("OUTCOME=duplicate")
	default:
		t.Fatal(err)
	}
}
